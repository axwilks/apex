create table countries (id uuid primary key default gen_random_uuid(), name_ru text not null, name_ky text, name_en text, sort int default 0); create table cities (id uuid primary key default gen_random_uuid(), country_id uuid references countries on delete restrict, name_ru text not null, name_ky text, name_en text, sort int default 0); create table platforms (id text primary key, name text not null, domains text[] not null, sort int default 0); create table venues (id uuid primary key default gen_random_uuid(), slug text unique not null check (slug ~ '^[a-z0-9-]+$'), name text not null, city_id uuid references cities, logo_url text, active boolean default true, sort int default 0, created_at timestamptz default now()); create table venue_links (venue_id uuid references venues on delete cascade, platform_id text references platforms, url text not null check (url like 'https://%'), enabled boolean default true, primary key (venue_id, platform_id));
