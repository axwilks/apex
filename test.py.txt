/* ========== 3. СОСТОЯНИЕ И ПОМОЩНИКИ ========== */
let lang = 'ru';
try { lang = localStorage.getItem('apex-lang') || 'ru'; } catch (e) {}
if (!UI[lang]) lang = 'ru';

let city = '';   // выбранный город ('' = не выбран)
let query = '';  // текст в строке поиска

const t = key => UI[lang][key];
const cityName = id => { const c = DATA.cities.find(c => c.id === id); return c ? (c.names[lang] || c.names.ru) : ''; };
const venueName = v => (v.names && v.names[lang]) || v.name;   // название как в базе, если нет перевода
const platformName = p => (p.names && p.names[lang]) || p.name;

// Создаёт элемент без innerHTML: текст вставляется как текст, подставить в него код нельзя
function el(tag, attrs = {}, ...kids) {
  const n = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs)) {
    if (k === 'class') n.className = v;
    else if (k.startsWith('on')) n.addEventListener(k.slice(2), v);
    else n.setAttribute(k, v);
  }
  n.append(...kids.flat().filter(Boolean));
  return n;
}

// Защита: пускаем только ссылки https://
function isSafeUrl(u) {
  try { return new URL(u).protocol === 'https:'; } catch (e) { return false; }
}

/* ========== 4. СТРАНИЦЫ ========== */
function homePage() {
  const list = el('div', { class: 'list' });

  function fillList() {
    list.replaceChildren();
    const term = query.trim().toLowerCase();
    if (!city && !term) { list.append(el('p', { class: 'hint' }, t('hint'))); return; }   // список не показываем заранее
    const found = DATA.venues.filter(v =>
      v.active && (!city || v.city === city) && (!term || venueName(v).toLowerCase().includes(term)));
    if (!found.length) { list.append(el('p', { class: 'hint' }, t('none'))); return; }
    for (const v of found) {
      list.append(el('a', { class: 'item', href: '#/r/' + v.slug },
        v.logo ? el('img', { src: v.logo, alt: '', onerror: e => e.target.remove() }) : null,   // нет фото: блока нет
        el('span', {}, venueName(v)),
        city ? null : el('small', {}, cityName(v.city))));
    }
  }

  const input = el('input', { type: 'search', placeholder: t('search'), 'aria-label': t('search'),
    autocomplete: 'off', value: query, oninput: e => { query = e.target.value; fillList(); } });

  const chips = el('div', { class: 'chips' }, DATA.cities.map(c =>
    el('button', { type: 'button', class: 'chip' + (city === c.id ? ' on' : ''),
      onclick: () => { city = city === c.id ? '' : c.id; render(); } }, c.names[lang] || c.names.ru)));

  fillList();
  return [el('h1', {}, t('greet')), el('p', { class: 'sub' }, t('sub')), chips, input, list];
}

function venuePage(slug) {
  const back = el('a', { class: 'back', href: '#/' }, '← ' + t('back'));
  const v = DATA.venues.find(v => v.slug === slug && v.active);
  if (!v) return [back, el('h1', {}, t('notFound'))];

  // кнопка есть только там, где есть ссылка и она безопасна
  const buttons = DATA.platforms
    .filter(p => v.links[p.id] && isSafeUrl(v.links[p.id]))
    .map(p => el('a', { class: 'btn', href: v.links[p.id], target: '_blank', rel: 'noopener noreferrer' }, platformName(p)));

  return [back,
    v.logo ? el('img', { class: 'venue-img', src: v.logo, alt: '', onerror: e => e.target.remove() }) : null,
    el('h1', {}, venueName(v)),
    el('p', { class: 'sub' }, t('venueText')),
    ...(buttons.length ? buttons : [el('p', { class: 'hint' }, t('noLinks'))])];
}

/* ========== 5. ШАПКА, ПОДВАЛ, МАРШРУТЫ ========== */
function render() {
  document.documentElement.lang = lang;

  document.getElementById('top').replaceChildren(
    el('a', { class: 'logo', href: '#/', 'aria-label': 'Apex' }, el('img', { src: 'logo.jpg', alt: 'Apex' })),
    el('div', { class: 'langs' }, Object.keys(UI).map(l =>
      el('button', { type: 'button', class: l === lang ? 'on' : '',
        onclick: () => { lang = l; try { localStorage.setItem('apex-lang', l); } catch (e) {} render(); } }, l.toUpperCase()))));

  const m = location.hash.match(/^#\/r\/([\w-]+)$/);   // адрес вида #/r/navat-bishkek
  document.getElementById('page').replaceChildren(...(m ? venuePage(m[1]) : homePage()));

  document.getElementById('foot').replaceChildren(
    el('a', { href: CONTACTS.whatsapp, target: '_blank', rel: 'noopener noreferrer' }, 'WhatsApp'),
    el('a', { href: CONTACTS.telegram, target: '_blank', rel: 'noopener noreferrer' }, 'Telegram'),
    el('span', { class: 'copy' }, '© ' + new Date().getFullYear() + ' Apex. ' + t('rights')));
}

window.addEventListener('hashchange', render);
render();
